```python
class IncrementalAnalyzer:
    def __init__(self, cache_dir: str = ".moss/cache"):
        self.cache = AnalysisCache(cache_dir)
        self.checksum = ChecksumCalculator()
        self.dependency_graph = None

    async def analyze(self, codebase: Codebase) -> AnalysisResult:
        # 1. Detect changed files
        changed_files = self.detect_changes(codebase)

        # 2. Determine impact set
        impact_set = self.calculate_impact_set(changed_files)

        # 3. Load cached results for unchanged files
        cached = self.cache.load_unchanged(codebase, impact_set)

        # 4. Analyze only changed + impacted files
        new_results = await self.analyze_files(impact_set)

        # 5. Merge and update cache
        merged = self.merge_results(cached, new_results)
        self.cache.update(impact_set, new_results)

        return merged

    def detect_changes(self, codebase: Codebase) -> List[Path]:
        """检测变更的文件"""
        changed = []
        for file in codebase.files:
            current_hash = self.checksum.calculate(file)
            cached_hash = self.cache.get_hash(file)
            if current_hash != cached_hash:
                changed.append(file)
        return changed
```

### 1.3 Parallel Processing Engine

```python
class ParallelAnalyzer:
    def __init__(self, max_workers: int = None):
        self.max_workers = max_workers or (os.cpu_count() or 4)
        self.executor = ProcessPoolExecutor(max_workers=self.max_workers)

    async def analyze_batch(self, files: List[Path]) -> List[FileAnalysis]:
        """并行分析多个文件"""
        loop = asyncio.get_event_loop()

        # 分批提交任务
        batch_size = max(1, len(files) // self.max_workers)
        batches = [files[i:i+batch_size] for i in range(0, len(files), batch_size)]

        futures = [
            loop.run_in_executor(self.executor, self._analyze_batch, batch)
            for batch in batches
        ]

        results = await asyncio.gather(*futures)
        return [r for batch in results for r in batch]
```

### 1.4 Multi-Level Caching Strategy

| Cache Level | Content | TTL | Storage |
|-------------|---------|-----|---------|
| L1 (Memory) | Parsed ASTs | 1 hour | RAM (100MB) |
| L2 (Disk) | Analysis results | 24 hours | SQLite (1GB) |
| L3 (Project) | Import graphs | 7 days | JSON files |
| L4 (Global) | LLM responses | 30 days | Shared cache |

```python
class MultiLevelCache:
    def __init__(self):
        self.l1 = LRUCache(maxsize=1000)  # memory
        self.l2 = DiskCache(".moss/cache")  # sqlite
        self.l3 = ProjectCache()  # json
        self.l4 = GlobalCache()  # redis/s3

    def get(self, key: str) -> Optional[Any]:
        # Try L1 → L2 → L3 → L4
        for cache in [self.l1, self.l2, self.l3, self.l4]:
            value = cache.get(key)
            if value is not None:
                # Promote to higher level
                self._promote(key, value, cache)
                return value
        return None
```

## 4. Machine Learning Enhancements

### 4.1 Refactoring Recommendation Engine

```python
class MLRefactoringRecommender:
    """
    基于历史数据推荐重构策略
    """

    def __init__(self, model_path: str = ".moss/models"):
        self.model = self._load_or_train_model(model_path)
        self.history = RefactoringHistory()

    def recommend(self, code_context: CodeContext) -> List[Recommendation]:
        """推荐重构策略"""
        # 特征提取
        features = self._extract_features(code_context)

        # 模型预测
        predictions = self.model.predict_proba([features])

        # 结合历史成功率
        recommendations = []
        for strategy, prob in zip(self.strategies, predictions[0]):
            historical_success = self.history.get_success_rate(strategy)
            score = prob * 0.7 + historical_success * 0.3

            recommendations.append(Recommendation(
                strategy=strategy,
                confidence=score,
                rationale=self._generate_rationale(strategy, features)
            ))

        return sorted(recommendations, key=lambda r: -r.confidence)

    def _extract_features(self, context: CodeContext) -> np.ndarray:
        """提取代码特征"""
        return np.array([
            context.lines_of_code,
            context.cyclomatic_complexity,
            context.cognitive_complexity,
            context.duplication_ratio,
            context.change_frequency,
            context.author_count,
            context.test_coverage,
        ])
```

### 4.2 Historical Learning

```python
class RefactoringHistory:
    """
    追踪重构历史，学习最优策略
    """

    def record(self, refactoring: RefactoringOperation, outcome: Outcome):
        """记录重构结果"""
        self.db.insert({
            'timestamp': datetime.now(),
            'strategy': refactoring.strategy,
            'code_context': refactoring.context.hash(),
            'success': outcome.success,
            'bugs_introduced': outcome.bugs,
            'review_time': outcome.review_time,
            'developer_satisfaction': outcome.satisfaction,
        })

    def get_success_rate(self, strategy: str) -> float:
        """获取策略历史成功率"""
        results = self.db.query(
            "SELECT success FROM refactorings WHERE strategy = ?",
            (strategy,)
        )
        if not results:
            return 0.5
        return sum(r['success'] for r in results) / len(results)

    def get_best_practices(self, code_pattern: str) -> List[str]:
        """获取特定代码模式的最佳实践"""
        similar = self.db.query(
            """SELECT strategy, AVG(success) as rate
               FROM refactorings
               WHERE context_hash LIKE ?
               GROUP BY strategy
               ORDER BY rate DESC""",
            (f"%{code_pattern}%",)
        )
        return [s['strategy'] for s in similar[:3]]
```

```python
class CodeReviewIntegration:
    """
    集成代码审查系统
    """

    def __init__(self, provider: str):  # github, gitlab, bitbucket
        self.provider = provider
        self.api = ReviewAPI(provider)

    def suggest_refactoring_in_pr(self, pr_id: str, issues: List[Issue]):
        """在PR中建议重构"""
        for issue in issues:
            self.api.create_comment(
                pr_id=pr_id,
                file_path=issue.file_path,
                line=issue.line,
                body=self._format_suggestion(issue)
            )

    def auto_apply_safe_fixes(self, pr_id: str) -> int:
        """自动应用安全的修复"""
        fixes = self._get_safe_fixes(pr_id)
        applied = 0
        for fix in fixes:
            if fix.confidence > 0.95:
                self.api.apply_suggestion(fix)
                applied += 1
        return applied
```

## Appendix: Technical Decisions

### ADR-001: Caching Strategy
**Decision:** Multi-level cache (L1 memory, L2 disk, L3 project, L4 global)  
**Rationale:** Balance between speed and persistence

### ADR-002: IDE Architecture
**Decision:** LSP-first, native plugins as wrappers  
**Rationale:** Single implementation serves all IDEs

### ADR-003: ML Approach
**Decision:** Lightweight models, on-device inference  
**Rationale:** Privacy, latency, no external dependencies
