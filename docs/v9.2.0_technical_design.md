### 2. Core Data Structures

```python
# Import Graph
@dataclass
class ImportGraph:
    """Directed graph of module dependencies"""
    graph: nx.DiGraph  # NetworkX directed graph
    
    def get_dependents(self, module: str) -> List[str]:
        """Get modules that import this module"""
        return list(self.graph.predecessors(module))
    
    def get_dependencies(self, module: str) -> List[str]:
        """Get modules imported by this module"""
        return list(self.graph.successors(module))
    
    def find_cycles(self) -> List[List[str]]:
        """Find circular dependencies"""
        return list(nx.simple_cycles(self.graph))

# Symbol Table
@dataclass
class SymbolTable:
    """Global symbol registry across all files"""
    symbols: Dict[str, SymbolInfo]  # name -> info
    
    def find_symbol(self, name: str) -> Optional[SymbolInfo]:
        return self.symbols.get(name)
    
    def find_usages(self, symbol: str) -> List[UsageLocation]:
        """Find all usages of a symbol across codebase"""
        symbol_info = self.symbols.get(symbol)
        return symbol_info.usages if symbol_info else []

@dataclass
class SymbolInfo:
    name: str
    kind: SymbolKind  # FUNCTION, CLASS, VARIABLE, etc.
    defined_in: Path
    line: int
    usages: List[UsageLocation]
    exports: bool  # Is this symbol exported (__all__ or public)

@dataclass
class UsageLocation:
    file: Path
    line: int
    column: int
    context: str  # Surrounding code for context

# Transformation Plan
@dataclass
class TransformationPlan:
    """Planned refactoring operations"""
    operations: List[RefactoringOperation]

class RefactoringOperation:
```

### 3. Algorithm: Function Move

```python
class MovePlanner:
    def plan_function_move(
        self,
        function: Function,
        source_module: Module,
        target_module: Module
    ) -> TransformationPlan:
        """
        Plan moving a function from source to target module.
        
        Steps:
        1. Identify all dependencies of the function
        2. Check if dependencies are available in target module
        3. Plan import updates in source module
        4. Plan import additions in target module
        5. Plan import updates in all dependent modules
        6. Generate rollback plan
        """
        operations = []
        
        # Step 1: Analyze function dependencies
        deps = self.analyze_dependencies(function)
        
        # Step 2: Check dependency availability
        unavailable = self.check_dependency_availability(deps, target_module)
        if unavailable:
            raise RefactoringError(
                f"Cannot move: dependencies {unavailable} not available in target"
            )
        
        # Step 3: Move function to target
        operations.append(RefactoringOperation(
            operation_type=OperationType.MOVE_FUNCTION,
            source_file=source_module.path,
            target_file=target_module.path,
            changes=[
                CodeChange(
                    action=Action.REMOVE,
                    location=Location(source_module.path, function.line_start, function.line_end),
                    content=function.source_code
                ),
                CodeChange(
                    action=Action.ADD,
                    location=Location(target_module.path, -1, -1),  # End of file
                    content=function.source_code
                )
            ]
        ))
        
        # Step 4: Update source module imports
        if function.is_public:
            operations.append(self.plan_export_removal(source_module, function.name))
        
        # Step 5: Update target module exports
        if function.is_public:
            operations.append(self.plan_export_addition(target_module, function.name))
        
        # Step 6: Update all modules that use this function
        usages = self.symbol_table.find_usages(function.name)
        for usage in usages:
            if usage.file != source_module.path and usage.file != target_module.path:
                operations.append(self.plan_import_update(
                    usage.file,
                    old_import=f"from {source_module.name} import {function.name}",
                    new_import=f"from {target_module.name} import {function.name}"
                ))
        
        return TransformationPlan(
            operations=operations,
            estimated_impact=self.estimate_impact(operations),
            rollback_steps=self.generate_rollback_steps(operations)
        )
```

### 4. Safety Validation

```python
class CrossFileSafetyValidator:
    def validate(self, plan: TransformationPlan, codebase: Codebase) -> ValidationResult:
        """
        Comprehensive validation before applying changes.
        """
        checks = [
            self.validate_syntax_consistency(plan),
            self.validate_import_consistency(plan, codebase),
            self.validate_reference_integrity(plan, codebase),
            self.validate_no_circular_imports(plan, codebase),
            self.validate_test_compatibility(plan, codebase),
        ]
        
        failures = [c for c in checks if not c.passed]
        
        return ValidationResult(
            passed=len(failures) == 0,
            checks=checks,
            failures=failures
        )
    
    def validate_syntax_consistency(self, plan: TransformationPlan) -> CheckResult:
        """
        Ensure all modified files remain syntactically valid.
        """
        for op in plan.operations:
            # Apply changes to temporary copy
            temp_content = self.apply_to_temp(op)
            try:
                ast.parse(temp_content)
            except SyntaxError as e:
                return CheckResult(
                    passed=False,
                    message=f"Syntax error in {op.source_file}: {e}"
                )
        
        return CheckResult(passed=True)
    
    def validate_import_consistency(self, plan: TransformationPlan, codebase: Codebase) -> CheckResult:
        """
        Ensure all imports remain valid after refactoring.
        """
        # Build import graph after changes
        temp_codebase = self.simulate_changes(plan, codebase)
        new_graph = ImportGraphBuilder().build(temp_codebase)
        
        # Check for broken imports
        broken = []
        for module in temp_codebase.modules:
            for imp in module.imports:
                if not self.import_exists(imp, new_graph):
                    broken.append(f"{module.name}: {imp}")
        
        if broken:
            return CheckResult(
                passed=False,
                message=f"Broken imports: {broken}"
            )
        
        return CheckResult(passed=True)
```

### 5. Transaction Management

```python
class TransactionManager:
    """
    Atomic transaction support for cross-file refactoring.
    """
    
    def __init__(self, git_integration: bool = True):
        self.changes: List[AppliedChange] = []
        self.git = GitIntegration() if git_integration else None
        self.backup_dir = Path(".moss/backups")
    
    def begin(self) -> Transaction:
        """Begin a new transaction"""
        if self.git:
            self.git.stash_changes()
        return Transaction(self)
    
    def apply(self, change: RefactoringOperation) -> None:
        """Apply a change and record it"""
        # Create backup
        backup_path = self.create_backup(change.source_file)
        
        # Apply change
        applied = self.apply_change(change)
        
        # Record
        self.changes.append(AppliedChange(
            operation=change,
            backup_path=backup_path,
            applied_at=datetime.now()
        ))
    
    def commit(self, message: str) -> CommitResult:
        """Commit all changes"""
        if self.git:
            self.git.commit_all(message)
        
        # Clear backups on successful commit
        self.clear_backups()
        
        return CommitResult(
            success=True,
            changes_count=len(self.changes),
            commit_hash=self.git.last_commit_hash if self.git else None
        )
    
    def rollback(self) -> RollbackResult:
        """Rollback all changes in transaction"""
        for change in reversed(self.changes):
            self.restore_backup(change.backup_path, change.operation.source_file)
        
        if self.git:
            self.git.reset_hard()
        
        return RollbackResult(success=True, restored_count=len(self.changes))
```

### 6. Performance Considerations

#### 6.1 Incremental Analysis

```python
class IncrementalAnalyzer:
    """
    Only re-analyze changed files.
    """
    
    def __init__(self, cache: AnalysisCache):
        self.cache = cache
        self.checksum = ChecksumCalculator()
    
    def analyze(self, codebase: Codebase) -> AnalysisResult:
        changed_files = []
        unchanged_files = []
        
        for file in codebase.files:
            current_hash = self.checksum.calculate(file)
            cached_hash = self.cache.get_hash(file)
            
            if current_hash != cached_hash:
                changed_files.append(file)
            else:
                unchanged_files.append(file)
        
        # Analyze only changed files
        new_results = self.analyze_files(changed_files)
        
        # Get cached results for unchanged files
        cached_results = self.cache.get_results(unchanged_files)
        
        # Merge and update affected dependencies
        merged = self.merge_results(new_results, cached_results)
        
        # Update cache
        self.cache.update(changed_files, new_results)
        
        return merged
```

#### 6.2 Parallel Processing

```python
class ParallelAnalyzer:
    """
    Analyze multiple files in parallel.
    """
    
    def __init__(self, max_workers: int = 4):
        self.executor = ProcessPoolExecutor(max_workers=max_workers)
    
    async def analyze_batch(self, files: List[Path]) -> List[FileAnalysis]:
        loop = asyncio.get_event_loop()
        
        # Submit all tasks
        futures = [
            loop.run_in_executor(self.executor, self.analyze_file, file)
            for file in files
        ]
        
        # Gather results
        results = await asyncio.gather(*futures)
        return results
```

### 7. API Design

```python
# Public API for v9.2.0

class MOSS:
    """Main entry point for MOSS v9.2.0"""
    
    def __init__(self, config: MOSSConfig):
        self.config = config
        self.refactor_engine = CrossFileRefactorEngine(config)
        self.analyzer = IncrementalAnalyzer(config.cache_dir)
    
    def analyze(self, path: Path) -> AnalysisReport:
        """Analyze codebase and return issues"""
        return self.analyzer.analyze(path)
    
    def refactor(
        self,
        request: RefactoringRequest,
        dry_run: bool = False
    ) -> RefactoringResult:
        """Execute refactoring"""
        return self.refactor_engine.execute(request, dry_run)
    
    def move_function(
        self,
        function_name: str,
        from_module: str,
        to_module: str,
        dry_run: bool = False
    ) -> RefactoringResult:
        """Convenience method for function move"""
        request = MoveFunctionRequest(
            function=function_name,
            source=from_module,
            target=to_module
        )
        return self.refactor(request, dry_run)
    
    def split_module(
        self,
        module_path: Path,
        split_strategy: SplitStrategy,
        dry_run: bool = False
    ) -> RefactoringResult:
        """Split a large module into package"""
        request = SplitModuleRequest(
            module=module_path,
            strategy=split_strategy
        )
        return self.refactor(request, dry_run)
```
